import java.text.MessageFormat;
import java.util.Scanner;

class Lab3_LastName_Group {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Введіть ціле число: ");
        int integerValue = scanner.nextInt();

        System.out.print("Введіть число з плаваючою точкою: ");
        double doubleValue = scanner.nextDouble();
        scanner.nextLine();

        System.out.print("Введіть строку: ");
        String stringValue = scanner.nextLine();

        System.out.print("Введіть логічне значення (true/false): ");
        boolean booleanValue = scanner.nextBoolean();

        System.out.println("\n========== 1. System.out.println ==========");

        System.out.println("Ціле число: " + integerValue);

        System.out.println("Дійсне число: " + doubleValue);

        System.out.println("Строка: " + stringValue + ", логічне значення: " + booleanValue);

        System.out.println("\n========== 2. MessageFormat ==========");

        System.out.println(MessageFormat.format(
                "Числа: ціле = {0}, дійсне = {1}", integerValue, doubleValue));

        System.out.println(MessageFormat.format(
                "Дані: {0} | {1} | {2}", integerValue, stringValue, booleanValue));

        System.out.println(MessageFormat.format(
                "Дійсне число у валютному форматі: {0,number,currency}", doubleValue));

        System.out.println(MessageFormat.format(
                "Дійсне число у відсотковому форматі: {0,number,percent}", doubleValue));

        System.out.println("\n========== 3. System.out.format ==========");

        System.out.format("Ціле число у десятковому форматі: %d%n", integerValue);

        System.out.format("Ціле число у шістнадцятковому форматі: %x%n", integerValue);

        System.out.format("Строка: %-15s | Дійсне: %10.2f | Логічне: %b%n",
                stringValue, doubleValue, booleanValue);

        scanner.close();
    }
}
