import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        System.out.println("ТИПИ ДАННИХ");
        System.out.println("byte");
        System.out.println("Розмір: 8 біт");
        System.out.println("Мінімальне значення:-128");
        System.out.println("Максимальне значення:127");

        System.out.println("short");
        System.out.println("Розмір: 2 байти ");
        System.out.println("Мінімальне значення:-32 768");
        System.out.println("Максимальне значення: 32 767");

        System.out.println("int");
        System.out.println("Розмір: 4 байти");
        System.out.println("Мінімальне значення:-2 147 483 648");
        System.out.println("Максимальне значення: 2 147 483 647");

        System.out.println("long");
        System.out.println("Розмір: 8 байтів");
        System.out.println("Мінімальне значення:-9 223 372 036 854 775 808");
        System.out.println("Максимальне значення:9 223 372 036 854 775 807");

        System.out.println("char");
        System.out.println("Розмір: 2 байти");
        System.out.println("Мінімальне значення: 0");
        System.out.println("Максимальне значення: 65535");

        System.out.println("boolean");
        System.out.println("Розмір: залежить від JVM (логічно 1 біт)");
        System.out.println("Значення: true або false");

        System.out.println("float");
        System.out.println("Розмір: 4 байти");
        System.out.println("Мінімальне значення: -3.4028235E38");
        System.out.println("Максимальне значення: 3.4028235E38");

        System.out.println("void");
        System.out.println("Розмір: немає");
        System.out.println("Значень немає, void означає, що метод нічого не повертає");

        System.out.println("double");
        System.out.println("Розмір: 8 байтів");
        System.out.println("Мінімальне значення: -1.7976931348623157E308");
        System.out.println("Максимальне значення: 1.7976931348623157E308");
    }
}
